# U-Boot environment for MTS Router devices
# This file is deployed alongside U-Boot and contains device-specific boot parameters

# ============================================================
# Common U-Boot environment variables
# ============================================================

# Boot arguments
bootargs_console="console=ttyS0,115200n8"
bootargs_root="rootwait rw"
bootargs_mts="mts.board=${mts_board} mts.os=${mts_os}"

# ============================================================
# Boot command
# ============================================================
bootcmd=run boot_detect; run boot_select; run boot_apply

# ============================================================
# Boot detection
# ============================================================
boot_detect=if env exists mts_board; then setenv mts_detected 1; else setenv mts_detected 0; fi

# ============================================================
# Boot selection
# ============================================================
boot_select=if test ${mts_detected} = 1; then run boot_main; else run boot_recovery; fi

# ============================================================
# Boot apply
# ============================================================
boot_apply=setenv bootargs "${bootargs_console} ${bootargs_root} ${bootargs_mts} ${extra_bootargs}"

# ============================================================
# Main boot
# ============================================================
boot_main=if mmc dev ${mmcdev}; then if fatload mmc ${mmcdev} ${loadaddr} boot/vmlinuz; then if fatload mmc ${mmcdev} ${fdtaddr} boot/${fdtfile}; then if fatload mmc ${mmcdev} ${rdaddr} boot/initrd.img; then booti ${loadaddr} ${rdaddr} ${fdtaddr}; else booti ${loadaddr} - ${fdtaddr}; fi; fi; fi; fi; run boot_recovery

# ============================================================
# Recovery boot
# ============================================================
boot_recovery=echo "Main boot failed, trying recovery..."; setenv bootdev usb; run boot_usb; run boot_net

# ============================================================
# USB boot
# ============================================================
boot_usb=if usb start; then if fatload usb 0 ${loadaddr} boot/vmlinuz; then if fatload usb 0 ${fdtaddr} boot/${fdtfile}; then booti ${loadaddr} - ${fdtaddr}; fi; fi; fi;

# ============================================================
# Network boot
# ============================================================
boot_net=if dhcp ${loadaddr} ${serverip}:${bootfile}; then if dhcp ${fdtaddr} ${serverip}:${fdtfile}; then booti ${loadaddr} - ${fdtaddr}; fi; fi
