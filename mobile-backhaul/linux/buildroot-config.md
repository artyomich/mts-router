## 2. board/mts/mb3000/post-build.sh

```bash
#!/bin/bash
# Post-build script for MTS-MB-3000

TARGET_DIR="$1"

# Create necessary directories
mkdir -p ${TARGET_DIR}/etc/init.d
mkdir -p ${TARGET_DIR}/var/log/mts
mkdir -p ${TARGET_DIR}/etc/mts
mkdir -p ${TARGET_DIR}/usr/share/mts

# Copy configuration files
cp ${OVERLAY_DIR}/mts-bh.conf ${TARGET_DIR}/etc/mts/
cp ${OVERLAY_DIR}/mts-ptp.conf ${TARGET_DIR}/etc/mts/
cp ${OVERLAY_DIR}/mts-mpls-tp.conf ${TARGET_DIR}/etc/mts/

# Copy init scripts
cp ${OVERLAY_DIR}/S01mts-bh-forwarder ${TARGET_DIR}/etc/init.d/
cp ${OVERLAY_DIR}/S02mts-ptp ${TARGET_DIR}/etc/init.d/
cp ${OVERLAY_DIR}/S03mts-mpls-tp ${TARGET_DIR}/etc/init.d/

# Set permissions
chmod +x ${TARGET_DIR}/etc/init.d/S01mts-bh-forwarder
chmod +x ${TARGET_DIR}/etc/init.d/S02mts-ptp
chmod +x ${TARGET_DIR}/etc/init.d/S03mts-mpls-tp

# Create symlinks
ln -sf /etc/init.d/S01mts-bh-forwarder ${TARGET_DIR}/etc/rc.d/S01mts-bh-forwarder
ln -sf /etc/init.d/S02mts-ptp ${TARGET_DIR}/etc/rc.d/S02mts-ptp
ln -sf /etc/init.d/S03mts-mpls-tp ${TARGET_DIR}/etc/rc.d/S03mts-mpls-tp

echo "Post-build complete for MTS-MB-3000"
```
